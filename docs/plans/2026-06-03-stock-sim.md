**Step 1: 创建 stock.ts**

```ts
export interface StockInfo {
  code: string        // 股票代码，如 '000001'
  name: string        // 股票名称，如 '平安银行'
  market: 'sh' | 'sz' // 上交所/深交所
  industry: string    // 行业
}

export interface StockQuote {
  code: string
  name: string
  price: number        // 当前价
  open: number         // 开盘价
  high: number         // 最高价
  low: number          // 最低价
  preClose: number     // 昨收
  volume: number       // 成交量（手）
  amount: number       // 成交额（万元）
  change: number       // 涨跌额
  changePercent: number // 涨跌幅（%）
  time: string         // 行情时间
}

export interface KlineData {
  time: string         // 日期
  open: number
  close: number
  high: number
  low: number
  volume: number
}

export type KlinePeriod = 'daily' | 'weekly' | 'monthly'

export interface MinutePoint {
  time: string         // HH:mm
  price: number
  volume: number
  avgPrice: number     // 均价
}

export interface DepthLevel {
  price: number
  volume: number       // 手
}

export interface DepthData {
  code: string
  bids: DepthLevel[]   // 买一 ~ 买五
  asks: DepthLevel[]   // 卖一 ~ 卖五
}

export interface OrderRequest {
  code: string
  type: 'buy' | 'sell'
  price: number
  quantity: number     // 股数（100 的倍数）
}

export interface Position {
  code: string
  name: string
  quantity: number     // 持仓股数
  avgCost: number      // 成本价
  currentPrice: number // 现价
  marketValue: number  // 市值
  profit: number       // 盈亏金额
  profitPercent: number // 盈亏比例
}

export interface TradeRecord {
  id: string
  code: string
  name: string
  type: 'buy' | 'sell'
  price: number
  quantity: number
  amount: number       // 成交金额
  fee: number          // 手续费
  time: string         // 成交时间
}

export interface AccountInfo {
  totalAssets: number    // 总资产
  availableCash: number  // 可用资金
  frozenCash: number     // 冻结资金
  marketValue: number    // 持仓市值
  totalProfit: number    // 累计盈亏
  totalProfitPercent: number
  initialCapital: number // 初始资金
}
```

**Step 2: 创建 data-source.ts**

```ts
import type { StockQuote, KlineData, KlinePeriod, MinutePoint, DepthData } from './stock'

export interface DataSourceAdapter {
  /** 数据源名称，展示在下拉切换列表中 */
  name: string
  /** 获取所有可选股票列表 */
  getStockList(): Promise<StockInfo[]>
  /** 获取单只股票行情 */
  getQuote(code: string): Promise<StockQuote>
  /** 获取 K 线数据 */
  getKline(code: string, period: KlinePeriod, count?: number): Promise<KlineData[]>
  /** 获取分时数据 */
  getMinuteLine(code: string): Promise<MinutePoint[]>
  /** 获取盘口深度 */
  getDepth(code: string): Promise<DepthData>
  /** 订阅实时行情推送 */
  subscribe(codes: string[], callback: (quote: StockQuote) => void): () => void
}

export type DataSourceKey = 'mock' | 'real'
```
